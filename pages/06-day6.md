// ขั้นที่ 2: เพิ่มเงื่อนไขส่วนลดสมาชิก 10%

function getDeliveryFee(totalPrice) {
  if (totalPrice >= 200) {
    return 0;
  } else if (totalPrice >= 100) {
    return 20;
  } else {
    return 35;
  }
}

console.log(getDeliveryFee(120)); // 20
console.log(getDeliveryFee(60));  // 35
