````md magic-move {lines: true}
let cart = [];

const orderButtons = document.querySelectorAll('.btn-order[data-name]');

const cartTotalEl = document.querySelector('#cart-total');
const discountMsgEl = document.querySelector('#discount-msg');

```

```js
function renderCart() {
  cartListEl.innerHTML = '';
  if (cart.length === 0) {
    cartListEl.innerHTML = '<li>ยังไม่มีรายการในตะกร้า</li>';
    cartTotalEl.textContent = '0'; discountMsgEl.textContent = ''; return;
  }
  let total = 0;
  cart.forEach((item, i) => {
    total += item.price;
    const li = document.createElement('li');
    li.textContent = `${i + 1}. ${item.name} — ${item.price} บาท`;
    cartListEl.appendChild(li);
  });
}
```

```js
// 3. เช็กส่วนลด (Logic จาก Day 6) และผูก Event คลิก
if (total >= 150) {
  cartTotalEl.textContent = total - 20;
  discountMsgEl.textContent = `ครบ 150 บาท! ลด 20 บาท (ปกติ ${total})`;
} else {
  cartTotalEl.textContent = total;
  discountMsgEl.textContent = `ซื้อเพิ่มอีก ${150 - total} บาท เพื่อลด 20 บาท`;
}
orderButtons.forEach((button) => {
  button.addEventListener('click', () => {
    cart.push({ name: button.dataset.name, price: Number(button.dataset.price) });
    renderCart();
  });
});
````

```js {lines:true}
cart.forEach((item, index) => {
  total += item.price;
  const li = document.createElement('li');
  li.textContent = `${index + 1}. ${item.name} — ${item.price} บาท `;
  const removeBtn = document.createElement('button');
  removeBtn.textContent = 'ลบ';
  removeBtn.style.marginLeft = '8px';
  removeBtn.addEventListener('click', () => {
    cart.splice(index, 1);
    renderCart();
  });
  li.appendChild(removeBtn);
  cartListEl.appendChild(li);
});
```
