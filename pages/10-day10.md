```php
<?php
require_once 'db.php';
$successMsg = '';

// เมื่อลูกค้ากดส่งฟอร์มสั่งออเดอร์ (POST)
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
  $menuId = (int) ($_POST['menu_id'] ?? 0);
  $customerName = trim($_POST['customer_name'] ?? '');
  $quantity = max(1, (int) ($_POST['quantity'] ?? 1));
}
```

```php
  // 1. ดึงราคาจริงจาก Database (ห้ามเชื่อราคาจาก Frontend!)
  $stmt = $pdo->prepare('SELECT name, price FROM menus WHERE id = ?');
  $stmt->execute([$menuId]);
  $menu = $stmt->fetch();

  if ($menu) {
    $totalPrice = $menu['price'] * $quantity;
    // 2. บันทึกออเดอร์ด้วย Prepared Statement
    $insert = $pdo->prepare(
      'INSERT INTO orders (menu_id, customer_name, quantity, total_price) VALUES (?, ?, ?, ?)'
    );
    $insert->execute([$menuId, $customerName, $quantity, $totalPrice]);
    $successMsg = "รับออเดอร์แล้ว! คุณ {$customerName} ({$totalPrice} บาท)";
  }
```

```php
// ดึงเมนูทั้งหมดพร้อมชื่อหมวดหมู่มาแสดงหน้าร้าน
$menus = $pdo->query('
  SELECT menus.*, categories.name AS category_name
  FROM menus
  JOIN categories ON menus.category_id = categories.id
  ORDER BY menus.id ASC
')->fetchAll();
?>
```

```php
// 1. เพิ่มเมนูใหม่ (CREATE)
if ($_SERVER['REQUEST_METHOD'] === 'POST' && ($_POST['action'] ?? '') === 'add') {
  $stmt = $pdo->prepare(
    'INSERT INTO menus (category_id, name, price, description) VALUES (?, ?, ?, ?)'
  );
  $stmt->execute([
    (int) $_POST['category_id'], trim($_POST['name']),
    (float) $_POST['price'], trim($_POST['description']),
  ]);
  header('Location: admin.php');
  exit;
}
```

```php
// 2. ลบเมนู (DELETE)
if ($_SERVER['REQUEST_METHOD'] === 'POST' && ($_POST['action'] ?? '') === 'delete') {
  $stmt = $pdo->prepare('DELETE FROM menus WHERE id = ?');
  $stmt->execute([(int) $_POST['menu_id']]);
  header('Location: admin.php');
  exit;
}
```

// 3. ดึงหมวดหมู่, เมนูทั้งหมด และออเดอร์ล่าสุด 20 รายการ (READ)
$categories = $pdo->query('SELECT * FROM categories ORDER BY id ASC')->fetchAll();
$menus = $pdo->query('
  SELECT menus.*, categories.name AS category_name
  FROM menus JOIN categories ON menus.category_id = categories.id
  ORDER BY menus.id DESC
')->fetchAll();
$orders = $pdo->query('
  SELECT orders.*, menus.name AS menu_name
  FROM orders JOIN menus ON orders.menu_id = menus.id
  ORDER BY orders.ordered_at DESC LIMIT 20
')->fetchAll();
