```yaml {all|2-5|6-9|10-13|14|all}{lines:true}
    build: .

  db:
    image: mariadb:11
    environment: { MARIADB_DATABASE: minicafe, MARIADB_USER: cafe_user, MARIADB_PASSWORD: cafe_password, MARIADB_ROOT_PASSWORD: rootpassword }
    volumes: ["db_data:/var/lib/mysql"]
  phpmyadmin:
    image: phpmyadmin:latest
    ports: ["8081:80"]
    environment: { PMA_HOST: db }
volumes: { db_data: }
```
